Subscribing requires a WebSocket connection and `subscribe` permission on the channel. Use the **Language** control to switch **JavaScript**, **React**, or **Go**.

REST cannot open a realtime subscription — publish with REST, receive with Socket.

<ExampleScope defaultGroup="Socket" groups={["Socket"]} languages={["javascript", "react", "go"]}>

## Basic subscribe

<CodeExamples group="Socket">

```javascript
import { QPub } from "@qpub/sdk";

const socket = new QPub.Socket({
  apiKey: process.env.QPUB_API_KEY,
});

socket.connection.on("connected", async () => {
  const channel = socket.channels.get("notifications");
  await channel.subscribe((message) => {
    console.log(message.data);
  });
});
```

```react
import { useCallback, useEffect } from "react";
import {
  SocketProvider,
  useChannel,
  useConnection,
  type Message,
} from "@qpub/sdk/react";

function App() {
  return (
    <SocketProvider options={{ apiKey: process.env.QPUB_API_KEY }}>
      <Inbox />
    </SocketProvider>
  );
}

function Inbox() {
  const { status } = useConnection();
  const { subscribe, unsubscribe, ready } = useChannel("notifications");

  const onMessage = useCallback((message: Message) => {
    console.log(message.data);
  }, []);

  useEffect(() => {
    if (!ready) return;
    subscribe(onMessage);
    return () => {
      unsubscribe(onMessage);
    };
  }, [ready, subscribe, unsubscribe, onMessage]);

  return <p>Status: {status}</p>;
}
```

```go
package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/qpubio/qpub-go/channel"
	"github.com/qpubio/qpub-go/qpub"
)

func main() {
	socket := qpub.NewSocket(qpub.WithAPIKey(os.Getenv("QPUB_API_KEY")))
	defer socket.Reset()

	ctx := context.Background()
	if err := socket.Connection.WaitUntilConnected(ctx); err != nil {
		log.Fatal(err)
	}

	ch := socket.Channels.Get("notifications")
	err := ch.Subscribe(ctx, func(m qpub.Message) {
		fmt.Println(m.Data)
	}, channel.SubscribeOptions{})
	if err != nil {
		log.Fatal(err)
	}
}
```

</CodeExamples>

<When lang="javascript">
`subscribe` returns a Promise that resolves when the server acknowledges the subscription.
</When>

<When lang="react">
`subscribe` resolves when the server acknowledges the subscription (via the hook).
</When>

<When lang="go">
`Subscribe` blocks until the server acknowledges the subscription or `ctx` is cancelled.
</When>

<When lang="javascript">
> Do not put a production API key secret in a public browser app. Use [token auth](/core/auth/token.mdx) instead.
</When>

<When lang="react">
Prefer a JWT or [token request](/core/auth/token.mdx) in `SocketProvider` options instead of embedding a production API key secret.
</When>

<When lang="go">
Use the API key only on trusted servers. For other clients, issue a [token request](/core/auth/token.mdx) or JWT instead.
</When>

## Event filter (SDK)

The client still subscribes to the whole channel on the server. The SDK delivers only matching `event` values to this handler.

<CodeExamples group="Socket">

```javascript
await channel.subscribe(
  (message) => console.log(message.data),
  { event: "order.created" }
);
```

```react
const { subscribe, unsubscribe, ready } = useChannel("notifications");

useEffect(() => {
  if (!ready) return;
  const onMessage = (message) => console.log(message.data);
  subscribe(onMessage, { event: "order.created" });
  return () => unsubscribe(onMessage);
}, [ready, subscribe, unsubscribe]);
```

```go
err := ch.Subscribe(ctx, func(m qpub.Message) {
	fmt.Println(m.Data)
}, channel.SubscribeOptions{Event: "order.created"})
```

</CodeExamples>

## Unsubscribe

<CodeExamples group="Socket">

```javascript
await channel.unsubscribe();
```

```react
const { unsubscribe } = useChannel("notifications");

// Remove a specific handler:
await unsubscribe(onMessage);

// Or omit the callback to clear handlers for this channel (SDK behavior).
await unsubscribe();
```

```go
if err := ch.Unsubscribe(ctx); err != nil {
	log.Fatal(err)
}
```

</CodeExamples>

## Production clients

Prefer a scoped JWT or token request so the browser never holds the API key secret:

<CodeExamples group="Socket">

```javascript
const socket = new QPub.Socket({
  tokenRequest, // from your backend
});
```

```react
function App({ tokenRequest }) {
  return (
    <SocketProvider options={{ tokenRequest }}>
      <Inbox />
    </SocketProvider>
  );
}
```

```go
import "github.com/qpubio/qpub-go/option"

socket := qpub.NewSocket(func(o *option.Option) {
	o.TokenRequest = tokenRequest // from your backend
})
defer socket.Reset()
```

</CodeExamples>

See [Token Auth](/core/auth/token.mdx).

## Protocol

Under the hood the client sends action `4` (subscribe) and receives `5` (subscribed) / `10` (message). See [Socket Protocol](/api/socket-protocol.mdx).
