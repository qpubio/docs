QPub’s official client library is `@qpub/sdk` for JavaScript and TypeScript (browsers, Node.js, Bun, and React).

## Install

```bash
npm install @qpub/sdk
# or
bun add @qpub/sdk
```

| Import | Use |
|--------|-----|
| `@qpub/sdk` | `QPub.Socket`, `QPub.Rest`, types |
| `@qpub/sdk/react` | React providers and hooks |
| CDN UMD | Global `QPub` in the browser |

## Socket client

Use for real-time subscribe and publish:

```javascript
import { QPub } from "@qpub/sdk";

const socket = new QPub.Socket({ apiKey: process.env.QPUB_API_KEY });

socket.connection.on("connected", async () => {
  const ch = socket.channels.get("news");
  await ch.subscribe((msg) => console.log(msg.data));
});
```

## REST client

Use for server publish, tokens, and queues:

```javascript
import { QPub } from "@qpub/sdk";

const rest = new QPub.Rest({ apiKey: process.env.QPUB_API_KEY });
await rest.channels.get("news").publish({ headline: "Ship it" });
```

## React

For browsers, pass a JWT or token request from your backend — not a secret API key in a public env var.

```tsx
import { useEffect, useCallback } from "react";
import {
  SocketProvider,
  useChannel,
  useConnection,
  type Message,
} from "@qpub/sdk/react";

function App({ tokenRequest }) {
  return (
    <SocketProvider options={{ tokenRequest }}>
      <Chat />
    </SocketProvider>
  );
}

function Chat() {
  const { status } = useConnection();
  const { publish, subscribe, unsubscribe, ready } = useChannel("chat");

  const onMessage = useCallback((message: Message) => {
    console.log(message.data);
  }, []);

  useEffect(() => {
    if (!ready) return;
    subscribe(onMessage);
    return () => {
      unsubscribe(onMessage);
    };
  }, [ready, subscribe, unsubscribe, onMessage]);

  return (
    <div>
      <p>Status: {status}</p>
      <button
        disabled={!ready}
        onClick={() => publish({ text: "Hello" }, { event: "chat" })}
      >
        Send
      </button>
    </div>
  );
}
```

See [Token Auth](/docs/core/auth/token) for issuing `tokenRequest` on the server.
